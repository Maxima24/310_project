# Real-hardware extras. Install on top of requirements.txt, and ONLY on the
# machine that has the hardware:
#
#   pip install -r requirements.txt -r requirements-hardware.txt
#
# --- Camera (--type camera --real) -----------------------------------------
#
# Verified working on Python 3.14 with opencv-python-headless 5.0.0 (wheel
# available, no source build needed) against a webcam, a local file, and an RTSP
# stream.
#
# If you land on a CPython release that OpenCV has not shipped a wheel for yet,
# pip will try a source build needing CMake and a C++ toolchain. Cheapest ways out:
#   1. Run the camera agent in Docker — agents/Dockerfile pins python:3.12-slim.
#   2. Make a 3.12/3.13 virtualenv just for the camera agent.
#
# Either way simulation mode is unaffected: cv2 is imported lazily, so an agent
# that never uses --real never touches OpenCV.
opencv-python-headless>=4.10

# --- Raspberry Pi GPIO (--type motion|door --real) -------------------------
#
# Then replace the marked stubs in security_agent/sensors.py — search for
# "HARDWARE STUB". Two commented lines per class.
gpiozero>=2.0
# Pin-factory backend. lgpio works on Pi 5 and Pi 4; use RPi.GPIO on older boards.
lgpio>=0.2.2.0; platform_machine == "aarch64" or platform_machine == "armv7l"

# --- Video evidence upload (--record-evidence with --s3-endpoint) -----------
# S3-compatible client. Imported lazily, so agents that never record clips do not
# need it installed.
boto3>=1.35
